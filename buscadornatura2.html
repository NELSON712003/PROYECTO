function doGet(e) { var callback = e.parameter.callback || 'callback'; var action = e.parameter.action || 'getData'; var sheetName = e.parameter.sheet || 'Hoja1'; var result = {}; try { // ========== PING - Prueba de conexion ========== if (action === 'ping') { result = { pong: true, message: 'Script funcionando correctamente', timestamp: new Date().toISOString() }; } // ========== GET DATA - Obtener datos de la hoja ========== else if (action === 'getData') { var rangeName = e.parameter.range || 'A:Z'; // Obtener la hoja activa (debe estar vinculada a este script) var ss = SpreadsheetApp.getActiveSpreadsheet(); if (!ss) { throw new Error('No se encontro una hoja de calculo vinculada a este script. Asegurate de abrir el script DESDE la hoja de calculo (Extensiones > Apps Script).'); } var sheet = ss.getSheetByName(sheetName); if (!sheet) { var nombres = ss.getSheets().map(function(s) { return s.getName(); }); throw new Error('Hoja "' + sheetName + '" no encontrada. Hojas disponibles: ' + nombres.join(', ')); } // Obtener datos del rango var range = sheet.getRange(rangeName); var rawData = range.getValues(); // Filtrar filas vacias al final var cleanData = []; for (var i = 0; i < rawData.length; i++) { var fila = rawData[i]; var tieneDatos = false; for (var j = 0; j < fila.length; j++) { if (fila[j] !== '' && fila[j] !== null && fila[j] !== undefined) { tieneDatos = true; break; } } if (tieneDatos) { cleanData.push(fila); } } result = { success: true, sheet: sheetName, range: rangeName, totalRows: cleanData.length, data: cleanData }; } // ========== DELETE - Eliminar fila ========== else if (action === 'delete') { var rowNum = parseInt(e.parameter.row); if (!rowNum || rowNum < 2) { throw new Error('Numero de fila invalido: ' + e.parameter.row); } var ss = SpreadsheetApp.getActiveSpreadsheet(); if (!ss) { throw new Error('No se encontro hoja de calculo vinculada'); } var sheet = ss.getSheetByName(sheetName); if (!sheet) { throw new Error('Hoja "' + sheetName + '" no encontrada'); } // Verificar que la fila existe if (rowNum > sheet.getLastRow()) { throw new Error('La fila ' + rowNum + ' no existe en la hoja'); } sheet.deleteRow(rowNum); result = { success: true, message: 'Fila ' + rowNum + ' eliminada', deletedRow: rowNum }; } // ========== ACCION DESCONOCIDA ========== else { throw new Error('Accion desconocida: ' + action); } } catch (err) { result = { error: err.toString(), action: action }; } // Devolver como JSONP - ESTO ES CRITICO var jsonString = JSON.stringify(result); var output = ContentService.createTextOutput(callback + '(' + jsonString + ');'); output.setMimeType(ContentService.MimeType.JAVASCRIPT); // Headers CORS (ayuda en algunos casos) output.setHeader('Access-Control-Allow-Origin', '*'); output.setHeader('Access-Control-Allow-Methods', 'GET, POST, OPTIONS'); return output; } // ============================================================ // doOptions - Para manejar preflight CORS // ============================================================ function doOptions(e) { var output = ContentService.createTextOutput(''); output.setMimeType(ContentService.MimeType.TEXT); output.setHeader('Access-Control-Allow-Origin', '*'); output.setHeader('Access-Control-Allow-Methods', 'GET, POST, OPTIONS'); output.setHeader('Access-Control-Allow-Headers', 'Content-Type'); return output; } // ============================================================ // doPost - Redirige a doGet (compatibilidad) // ============================================================ function doPost(e) { return doGet(e); } // ============================================================ // FUNCIONES DE PRUEBA (ejecutar desde el editor) // ============================================================ function testPing() { var e = { parameter: { action: 'ping', callback: 'test' } }; var result = doGet(e); Logger.log(result.getContent()); } function testGetData() { var e = { parameter: { action: 'getData', sheet: 'Hoja1', range: 'A:Z', callback: 'test' } }; var result = doGet(e); Logger.log(result.getContent()); } function testDelete() { var e = { parameter: { action: 'delete', sheet: 'Hoja1', row: '5', callback: 'test' } }; var result = doGet(e); Logger.log(result.getContent()); }
